import { createClient } from '@supabase/supabase-js' const supabaseUrl = import.meta.env.VITE_SUPABASE_URL const supabasePublishableKey = import.meta.env.VITE_SUPABASE_PUBLISHABLE_KEY if (!supabaseUrl || !supabasePublishableKey) { throw new Error('Brak VITE_SUPABASE_URL lub VITE_SUPABASE_PUBLISHABLE_KEY w pliku .env.local') } export const supabase = createClient(supabaseUrl, supabasePublishableKey, { auth: { persistSession: true, autoRefreshToken: true, detectSessionInUrl: true, }, })
